// === EMA'LAR (200, 377, 610) ===
ema200 = ta.ema(close, 200)
ema377 = ta.ema(close, 377)
ema610 = ta.ema(close, 610)

// EMA Uzaklıkları
dist_ema_200 = (close - ema200) / ema200
dist_ema_377 = (close - ema377) / ema377
dist_ema_610 = (close - ema610) / ema610

// EMA Gap'ler
ema_200_377_gap = math.abs(ema200 - ema377) / close
ema_377_610_gap = math.abs(ema377 - ema610) / close
ema_all_gap = math.abs(ema200 - ema610) / close

// EMA Yakınsama (%2'den az fark)
emas_converging = ema_all_gap < 0.02 ? 1 : 0

// EMA Destek/Direnç (fiyat EMA'ya %2'den yakın)
near_threshold = 0.02
ema_200_support = dist_ema_200 > 0 and dist_ema_200 < near_threshold ? 1 : 0
ema_200_resistance = dist_ema_200 < 0 and math.abs(dist_ema_200) < near_threshold ? 1 : 0
ema_377_support = dist_ema_377 > 0 and dist_ema_377 < near_threshold ? 1 : 0
ema_377_resistance = dist_ema_377 < 0 and math.abs(dist_ema_377) < near_threshold ? 1 : 0
ema_610_support = dist_ema_610 > 0 and dist_ema_610 < near_threshold ? 1 : 0
ema_610_resistance = dist_ema_610 < 0 and math.abs(dist_ema_610) < near_threshold ? 1 : 0

// Destek Gücü Skoru
support_strength = ema_200_support * 1 + ema_377_support * 2 + ema_610_support * 3 + emas_converging * 2
resistance_strength = ema_200_resistance * 1 + ema_377_resistance * 2 + ema_610_resistance * 3 + emas_converging * 2
near_ema_support = support_strength > 0 ? 1 : 0
near_ema_resistance = resistance_strength > 0 ? 1 : 0
super_support_zone = emas_converging == 1 and support_strength >= 3 ? 1 : 0

// EMA Durumları
above_ema_200 = close > ema200 ? 1 : 0
above_ema_377 = close > ema377 ? 1 : 0
above_ema_610 = close > ema610 ? 1 : 0
above_all_emas = close > ema200 and close > ema377 and close > ema610 ? 1 : 0
ema_stack = ema200 > ema377 and ema377 > ema610 ? 1 : 0

// === STOCHASTIC RSI ===
smoothK = 3
smoothD = 3
lengthRSI = 14
lengthStoch = 14
rsi1 = ta.rsi(close, lengthRSI)
stoch_k = ta.sma(ta.stoch(rsi1, rsi1, rsi1, lengthStoch), smoothK)
stoch_d = ta.sma(stoch_k, smoothD)

// === POI - DESTEK/DİRENÇ ===
swing_length = 10
swing_high = ta.highest(high, swing_length * 2 + 1)
swing_low = ta.lowest(low, swing_length * 2 + 1)
dist_to_resistance = (swing_high - close) / close
dist_to_support = (close - swing_low) / close
near_support = dist_to_support < 0.02 ? 1 : 0
near_resistance = dist_to_resistance < 0.02 ? 1 : 0

// === TRAILING STOP UZAKLIĞI ===
// NOT: xATRTrailingStop senin UT Bot kodunda zaten tanımlı olmalı
// Eğer farklı bir isimle tanımlıysan aşağıdaki satırı değiştir
dist_to_stop = (close - xATRTrailingStop) / close
